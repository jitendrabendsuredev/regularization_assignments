House Price Prediction

Features:
Area Bedrooms HouseAge DistanceFromCity LocationScore

Target: Price

⬇ Download CSV
  1. Load and inspect CSV.
  2. Create X/y and 80:20 split.
  3. Scale features.
  4. Train Linear, Ridge and Lasso.
  5. Compare MAE, RMSE, train/test R².
  6. Tune alpha using CV.
  7. Compare coefficients and write conclusion.

Complete Python Solution

import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LinearRegression, Ridge, Lasso
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score

df = pd.read_csv("01_house_price.csv")
print(df.head())
print(df.isnull().sum())

X = df[['Area', 'Bedrooms', 'HouseAge', 'DistanceFromCity', 'LocationScore']]
y = df["Price"]

X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.20, random_state=42
)

models = {
 "Linear": Pipeline([("scale", StandardScaler()),("model", LinearRegression())]),
 "Ridge": Pipeline([("scale", StandardScaler()),("model", Ridge(alpha=1.0))]),
 "Lasso": Pipeline([("scale", StandardScaler()),("model", Lasso(alpha=0.1, max_iter=10000))])
}

for name, model in models.items():
    model.fit(X_train, y_train)
    pred = model.predict(X_test)
    print("\n", name)
    print("Train R2:", model.score(X_train, y_train))
    print("Test R2:", r2_score(y_test, pred))
    print("MAE:", mean_absolute_error(y_test, pred))
    print("RMSE:", np.sqrt(mean_squared_error(y_test, pred)))

# Ridge alpha tuning
ridge = Pipeline([("scale",StandardScaler()),("model",Ridge())])
grid = GridSearchCV(
    ridge,
    {"model__alpha":[0.01,0.1,1,10,100]},
    cv=5, scoring="r2"
)
grid.fit(X_train,y_train)
print("Best Ridge Alpha:",grid.best_params_)
print("Ridge Test R2:",grid.score(X_test,y_test))

# Lasso alpha tuning
lasso = Pipeline([("scale",StandardScaler()),("model",Lasso(max_iter=10000))])
lgrid = GridSearchCV(
    lasso,
    {"model__alpha":[0.01,0.1,1,10,100]},
    cv=5, scoring="r2"
)
lgrid.fit(X_train,y_train)
print("Best Lasso Alpha:",lgrid.best_params_)
print("Lasso Test R2:",lgrid.score(X_test,y_test))

# Coefficient comparison
print("\nRidge coefficients:")
print(dict(zip(X.columns,grid.best_estimator_.named_steps["model"].coef_)))
print("\nLasso coefficients:")
print(dict(zip(X.columns,lgrid.best_estimator_.named_steps["model"].coef_)))

import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LinearRegression, Ridge, Lasso
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score

df = pd.read_csv("10_delivery_cost.csv")
print(df.head())
print(df.isnull().sum())

X = df[['DistanceKM', 'WeightKG', 'TrafficScore', 'FuelPrice', 'DeliveryTime']]
y = df["DeliveryCost"]

X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.20, random_state=42
)

models = {
 "Linear": Pipeline([("scale", StandardScaler()),("model", LinearRegression())]),
 "Ridge": Pipeline([("scale", StandardScaler()),("model", Ridge(alpha=1.0))]),
 "Lasso": Pipeline([("scale", StandardScaler()),("model", Lasso(alpha=0.1, max_iter=10000))])
}

for name, model in models.items():
    model.fit(X_train, y_train)
    pred = model.predict(X_test)
    print("\n", name)
    print("Train R2:", model.score(X_train, y_train))
    print("Test R2:", r2_score(y_test, pred))
    print("MAE:", mean_absolute_error(y_test, pred))
    print("RMSE:", np.sqrt(mean_squared_error(y_test, pred)))

# Ridge alpha tuning
ridge = Pipeline([("scale",StandardScaler()),("model",Ridge())])
grid = GridSearchCV(
    ridge,
    {"model__alpha":[0.01,0.1,1,10,100]},
    cv=5, scoring="r2"
)
grid.fit(X_train,y_train)
print("Best Ridge Alpha:",grid.best_params_)
print("Ridge Test R2:",grid.score(X_test,y_test))

# Lasso alpha tuning
lasso = Pipeline([("scale",StandardScaler()),("model",Lasso(max_iter=10000))])
lgrid = GridSearchCV(
    lasso,
    {"model__alpha":[0.01,0.1,1,10,100]},
    cv=5, scoring="r2"
)
lgrid.fit(X_train,y_train)
print("Best Lasso Alpha:",lgrid.best_params_)
print("Lasso Test R2:",lgrid.score(X_test,y_test))

# Coefficient comparison
print("\nRidge coefficients:")
print(dict(zip(X.columns,grid.best_estimator_.named_steps["model"].coef_)))
print("\nLasso coefficients:")
print(dict(zip(X.columns,lgrid.best_estimator_.named_steps["model"].coef_)))
